14)namespace Ejercicio14
{
    internal class Program
    {
        static void Main(string[] args)
        {
            try
            {
                Console.Write("Ingrese una frase: ");
                string frase = Console.ReadLine();


                if (frase.Length == 0)
                {
                    throw new ArgumentException("La frase no puede estar vacía.");
                }


                Console.WriteLine("Cantidad de caracteres: " + frase.Length);


                string[] palabras = frase.Split(' ');
                Console.WriteLine(" Cantidad de palabras: " + palabras.Length);


                int vocales = 0;
                foreach (char c in frase.ToLower())
                {
                    string letra = c.ToString();
                    if (letra == "a" || letra == "e" || letra == "i" || letra == "o" || letra == "u")
                    {
                        vocales = vocales + 1;
                    }
                }
                Console.WriteLine(" Cantidad de vocales: " + vocales);




                Console.Write("\nIngrese una posición numérica: ");
                int posicion = Convert.ToInt32(Console.ReadLine());
                Console.WriteLine(" Letra en esa posición: " + frase[posicion]);




                Console.Write("\nIngrese un número para dividir 100: ");
                int divisor = Convert.ToInt32(Console.ReadLine());
                Console.WriteLine("100 dividido " + divisor + " es: " + (100 / divisor));
            }
            catch (ArgumentException error)
            {
                Console.WriteLine("Error: " + error.Message);
            }
            catch (FormatException)
            {
                Console.WriteLine("Error: Pusiste letras en un campo que requería números.");
            }
            catch (IndexOutOfRangeException)
            {
                Console.WriteLine("Error: La posición indicada no existe en la frase.");
            }
            catch (DivideByZeroException)
            {
                Console.WriteLine("Error: No se permite la división por cero.");
            }


        }
    }
}
